Atom ::= var
       | int
       | bool
       | "(" Exp ")"

FExp' ::= (* empty *)
        | Atom FExp'

FExp ::= Atom FExp'

LExp ::= "if" Exp "then" Exp "else" Exp
       | FExp

Exp2' ::= (* empty *)
        | "**" LExp Exp2'

Exp2 ::= LExp Exp2'

Exp1' ::= (* empty *)
        | "*" Exp2 Exp1'
        | "/" Exp2 Exp1'

Exp1 ::= Exp2 Exp1'

Exp0' ::= (* empty *)
        | "+" Exp1 Exp0'
        | "-" Exp1 Exp0'

Exp0 ::= Exp1 Exp0'

ExpEq' ::= (* empty *)
        | "==" Exp0 ExpEq'

ExpEq ::= Exp0 ExpEq'

Exp  ::= ExpEq
